     4→     4→
     5→     5→   - Creates/joins a room on the party server (party-server/)

   184→   184→  function parseYouTubeId(url) {
   185→   185→    if (typeof url !== 'string') return null;
   186→   186→    const patterns = [
   187→   187→      /youtu\.be\/([\w-]{11})(?:[?&#]|$)/,
   188→   188→      /[?&]v=([\w-]{11})(?:[&]|$)/,
   189→   189→      /youtube\.com\/embed\/([\w-]{11})(?:[?&#]|$)/,
   190→   190→      /youtube\.com\/shorts\/([\w-]{11})(?:[?&#]|$)/,
   191→   191→    ];
   192→   192→    for (const p of patterns) {
   193→   193→      const m = url.match(p);
   194→   194→      if (m) return m[1];
   195→   195→    }
   196→   196→    return null;
   197→   197→  }

   199→   199→  function detectMediaType(url) {

   201→   201→    if (!u) return 'unknown';
   202→   202→    if (parseYouTubeId(u)) return 'youtube';
   203→   203→    if (/\.mp4([?#]|$)/i.test(u)) return 'mp4';
   204→   204→    return 'unknown';
   205→   205→  }

   310→   310→  function onState(room) {

   324→   324→
   325→   325→    if (room.media && room.media.url) {
   326→   326→      const roomChanged = state.roomMediaUrl !== room.media.url;
   327→   327→      if (roomChanged) {
   328→   328→        state.roomMediaUrl = room.media.url;
   329→   329→        state.localOverride = false;
   330→   330→        applyMedia(room.media, { time: state.lastState.time, paused: state.lastState.paused, force: true });
   331→   331→      } else if (!state.localOverride && !state.isHost) {
   332→   332→        syncPlayback(); /* drift correction */
   333→   333→      }
   334→   334→    } else if (state.roomMediaUrl || state.currentMedia) {
   335→   335→      state.roomMediaUrl = null;
   336→   336→      state.localOverride = false;
   337→   337→      resetStage();
   338→   338→    }
   339→   339→    updateSyncPill();
   340→   340→  }

   369→   369→  function onPlaybackEvent(kind, d) {
   370→   370→    if (state.lastState) {
   371→   371→      state.lastState.receivedAt = performance.now();
   372→   372→      state.lastState.time = Number(d && d.time) || 0;
   373→   373→      if (kind === 'play') state.lastState.paused = false;
   374→   374→      if (kind === 'pause') state.lastState.paused = true;
   375→   375→    }
   376→   376→    if (state.isHost) { updateSyncPill(); return; } /* host drives, never follows */
   377→   377→    if (state.localOverride) return;
   378→   378→    if (kind === 'seek') { hardSeek(Number(d && d.time) || 0); }
   379→   379→    else { syncPlayback(); }
   380→   380→    updateSyncPill();
   381→   381→  }

   512→   512→      document.head.appendChild(s);
   513→   513→      setTimeout(() => {
   514→   514→        if (!(window.YT && window.YT.Player)) { ytApiPromise = null; reject(new Error('YouTube API timeout')); }
   515→   515→      }, 12000);
   516→   516→    });

   520→   520→  async function loadYouTube(media, opts) {
   521→   521→    const videoId = parseYouTubeId(media.url);
   522→   522→    if (!videoId) {
   523→   523→      /* safety net if the server type disagrees with the URL */
   524→   524→      applyMedia({ type: 'unknown', url: media.url, title: media.title }, opts);
   525→   525→      return;
   526→   526→    }
   527→   527→    try {
   528→   528→      await ensureYouTubeAPI();
   529→   529→    } catch (err) {
   530→   530→      toast("Couldn't load the YouTube player.", 'error');
   531→   531→      return;
   532→   532→    }
   533→   533→    if (!ytPlayer) {
   534→   534→      ytPendingApply = { videoId, opts };
   535→   535→      ytPlayer = new YT.Player('yt-player', {
   536→   536→        width: '100%',
   537→   537→        height: '100%',
   538→   538→        videoId,
   539→   539→        playerVars: {
   540→   540→          controls: 1,
   541→   541→          rel: 0,
   542→   542→          modestbranding: 1,
   543→   543→          playsinline: 1,
   544→   544→        },
   545→   545→        events: {
   546→   546→          onReady: () => {
   547→   547→            const pending = ytPendingApply;
   548→   548→            ytPendingApply = null;
   549→   549→            if (pending) applyYouTubeVideo(pending.videoId, pending.opts);
   550→   550→          },
   551→   551→          onStateChange: onYouTubeStateChange,
   552→   552→          onError: () => toast("Couldn't load this YouTube video.", 'error'),
   553→   553→        },

   555→   555→    } else {
   556→   556→      applyYouTubeVideo(videoId, opts);
   557→   557→    }
   558→   558→  }

   560→   560→  function applyYouTubeVideo(videoId, opts) {
   561→   561→    if (!ytPlayer) return;
   562→   562→    const t = Math.max(0, Number(opts.time) || 0);
   563→   563→    let currentVideoId = null;
   564→   564→    try { currentVideoId = ytPlayer.getVideoData() && ytPlayer.getVideoData().video_id; } catch (err) {}
   565→   565→
   566→   566→    if (currentVideoId !== videoId) {
   567→   567→      /* per spec: paused -> cueVideoById, playing -> loadVideoById */
   568→   568→      if (opts.paused) {
   569→   569→        try { ytPlayer.cueVideoById({ videoId, startSeconds: t }); } catch (err) {}
   570→   570→      } else {
   571→   571→        try { ytPlayer.loadVideoById({ videoId, startSeconds: t }); } catch (err) {}
   572→   572→      }
   573→   573→    } else {
   574→   574→      try {
   575→   575→        const cur = ytPlayer.getCurrentTime() || 0;
   576→   576→        if (Math.abs(cur - t) > DRIFT_TOLERANCE) ytPlayer.seekTo(t, true);
   577→   577→      } catch (err) {}
   578→   578→    }
   579→   579→    /* then apply pause/play accordingly */
   580→   580→    try { if (opts.paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo(); } catch (err) {}
   581→   581→  }

   592→   592→    }

   605→   605→  function syncPlayback() {

   607→   607→    const paused = state.lastState.paused;
   608→   608→    const target = targetTime();
   609→   609→
   610→   610→    if (state.currentMedia.type === 'mp4') {
   611→   611→      const v = els.mp4;
   612→   612→      try { if (Math.abs(v.currentTime - target) > DRIFT_TOLERANCE) v.currentTime = target; } catch (err) {}
   613→   613→      if (paused && !v.paused) v.pause();
   614→   614→      if (!paused && v.paused) v.play().catch(showTapOverlay);
   615→   615→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   616→   616→      try {
   617→   617→        const cur = ytPlayer.getCurrentTime() || 0;
   618→   618→        if (Math.abs(cur - target) > DRIFT_TOLERANCE) ytPlayer.seekTo(target, true);
   619→   619→        if (paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo();
   620→   620→      } catch (err) {}
   621→   621→    }
   622→   622→  }

   642→   642→  setInterval(() => {
   643→   643→    /* host: YouTube has no "seeked" event — detect time jumps
   644→   644→       (both while playing AND while paused) */
   645→   645→    if (state.isHost && state.currentMedia && state.currentMedia.type === 'youtube' && ytPlayer) {
   646→   646→      try {
   647→   647→        const cur = ytPlayer.getCurrentTime();

   649→   649→        if (cur != null && st !== window.YT.PlayerState.UNSTARTED) {
   650→   650→          if (lastHostYtTime != null && Math.abs(cur - lastHostYtTime) > 1.8) {
   651→   651→            emitHostPlayback('seek', cur);
   652→   652→          }
   653→   653→          lastHostYtTime = cur;
   654→   654→        } else {
   655→   655→          lastHostYtTime = null;
   656→   656→        }
   657→   657→      } catch (err) { lastHostYtTime = null; }
   658→   658→    } else {
   659→   659→      lastHostYtTime = null;
   660→   660→    }
   661→   661→
   662→   662→    /* non-host: follow the room */
   663→   663→    if (!state.isHost) syncPlayback();

   670→   670→    if (!state.currentMedia) { setSync('waiting', 'Waiting for video…'); return; }

   674→   674→
   675→   675→    let localTime = null;

   677→   677→    if (state.currentMedia.type === 'mp4') {
   678→   678→      localTime = els.mp4.currentTime || 0;

   682→   682→    }
   683→   683→    if (localTime == null) { setSync('waiting', 'Syncing…'); return; }
   684→   684→
   685→   685→    const drift = Math.abs(localTime - targetTime());
   686→   686→    const pausedMatch = !!ls.paused === !!localPaused;
   687→   687→    if (pausedMatch && drift <= DRIFT_TOLERANCE) {
   688→   688→      setSync('synced', ls.paused ? 'Paused · in sync (' + fmtClock(localTime) + ')' : 'In sync · ' + fmtClock(localTime));
   689→   689→    } else {

   691→   691→    }

   902→   902→
