   162→function initProviderTabs() {
   163→  const tabs = $$('.provider-tab');
   164→  if (!tabs.length) return;
   165→
   166→  let activeTab = tabs.find((tab) => tab.classList.contains('active')) || tabs[0];
   167→
   168→  tabs.forEach((tab) => {
   169→    tab.addEventListener('click', () => {
   170→      if (tab.classList.contains('active')) return;
   171→      activeTab = tab;
   172→      tabs.forEach((other) => {
   173→        const isActive = other === tab;
   174→        other.classList.toggle('active', isActive);
   175→        other.setAttribute('aria-selected', String(isActive));
   176→      });
   177→      setProviderBrand(tab.dataset.provider);
   178→      loadProviderContent(tab.dataset.provider, tab.dataset.name);
   179→    });
   180→
   181→    // Hovering a tab previews that brand instantly: the ambient
   182→    // wash glides over, the tab scales + glows, and the header
   183→    // text swaps to the hovered brand ("Only on DISNEY+", ...).
   184→    tab.addEventListener('mouseenter', () => {
   185→      setProviderBrand(tab.dataset.provider);
   186→      setProviderHeading(tab.dataset.name);
   187→    });
   188→  });
   189→
   190→  const strip = $('#provider-tabs');
   191→  if (strip) {
   192→    // Leaving the tab strip restores the ACTIVE brand + header.
   193→    strip.addEventListener('mouseleave', () => {
   194→      setProviderBrand(activeTab ? activeTab.dataset.provider : '8');
   195→      setProviderHeading(activeTab ? activeTab.dataset.name : 'Netflix');
   196→    });
   197→  }
   198→
   199→  // Default active tab: Netflix
   200→  setProviderBrand('8');
   201→  loadProviderContent('8', 'Netflix');
   202→}

   209→/** Fetch a provider's movies AND TV shows, merge them into one
   210→    mixed rail (interleaved, de-duplicated, 24 titles) and render
   211→    it with the standard card component. */
   212→async function loadProviderMixedRail(providerId) {
   213→  const row = $('#provider-row');
   214→  if (!row) return;
   215→  showSkeletons(row, 8, 'card');
   216→  try {
   217→    const [moviesRes, tvRes] = await Promise.allSettled([
   218→      discoverByProvider(providerId),
   219→      discoverTVByProvider(providerId)
   220→    ]);
   221→
   222→    const movies = moviesRes.status === 'fulfilled' ? (moviesRes.value.results || []) : [];
   223→    const shows = tvRes.status === 'fulfilled' ? (tvRes.value.results || []) : [];
   224→
   225→    // Tag the media type explicitly — discover endpoints return
   226→    // bare results, and movie/tv id namespaces can collide.
   227→    movies.forEach((m) => { m.media_type = 'movie'; });
   228→    shows.forEach((s) => { s.media_type = 'tv'; });
   229→
   230→    // Interleave movie/TV/movie/TV so both media are represented
   231→    // along the whole rail, then drop numeric id collisions.
   232→    const seen = new Set();
   233→    const mixed = [];
   234→    const maxLen = Math.max(movies.length, shows.length);
   235→    for (let i = 0; i < maxLen; i++) {
   236→      [movies[i], shows[i]].forEach((item) => {
   237→        if (!item || !item.poster_path) return;
   238→        const key = `${item.media_type}-${item.id}`;
   239→        if (seen.has(key)) return;
   240→        seen.add(key);
   241→        mixed.push(item);
   242→      });
   243→    }
   244→
   245→    const items = mixed.slice(0, 24); // 20+ titles per provider
   246→    if (!items.length) {
   247→      showEmptyState(row, 'No titles available for this service right now.');
   248→      return;
   249→    }
   250→
   251→    row.innerHTML = '';
   252→    const fragment = document.createDocumentFragment();
   253→    items.forEach((item) => fragment.appendChild(createMovieCard(item)));
   254→    row.appendChild(fragment);
   255→  } catch (error) {
   256→    showErrorState(row, error.message, () => loadProviderMixedRail(providerId));
   257→  }
   258→}

   270→  const row = $('#tv-upcoming-row');
   271→  if (!row) return;
   272→  showSkeletons(row, 8, 'card-wide');
   273→  try {
   274→    const [onAir, airingToday] = await Promise.allSettled([
   275→      getOnTheAirTV(),
   276→      getAiringTodayTV()
   277→    ]);
   278→    const onAirItems = onAir.status === 'fulfilled' ? (onAir.value.results || []) : [];
   279→    const airingItems = airingToday.status === 'fulfilled' ? (airingToday.value.results || []) : [];
   280→    if (!onAirItems.length && !airingItems.length) {
   281→      throw new Error('No TV schedule data available.');
   282→    }
   283→
   284→    // Shows with an episode airing today come first, then the rest.
   285→    const seen = new Set();
   286→    const items = [...airingItems, ...onAirItems]
   287→      .filter((show) => {
   288→        if (!show || !show.backdrop_path || seen.has(show.id)) return false;
   289→        seen.add(show.id);
   290→        return true;
   291→      })
   292→      .slice(0, CONFIG.cardsPerRow);
   293→
   294→    if (!items.length) {
   295→      showEmptyState(row, 'No upcoming TV shows available right now.');
   296→      return;
   297→    }
   298→
   299→    // List endpoints don't carry episode data — pull each show's
   300→    // brief details (next/last episode) so cards can show the
   301→    // "New episode" / "New season" flags and "S1·E3" info.
   302→    // Requests are staggered to stay well inside TMDB rate limits.
   303→    const enriched = await Promise.all(items.map(async (show, index) => {
   304→      try {
   305→        await new Promise((resolve) => setTimeout(resolve, index * 60));
   306→        const detail = await getTVBrief(show.id);
   307→        return {
   308→          ...show,
   309→          next_episode_to_air: detail.next_episode_to_air || null,
   310→          last_episode_to_air: detail.last_episode_to_air || null
   311→        };
   312→      } catch (error) {
   313→        return show; // episode info is optional decoration
   314→      }
   315→    }));
   316→
   317→    row.innerHTML = '';
   318→    const fragment = document.createDocumentFragment();
   319→    enriched.forEach((show) => {
   320→      const { badge, metaNote } = describeTvEpisode(show);
   321→      fragment.appendChild(createMovieCard(show, { wide: true, badge, metaNote }));
   322→    });
   323→    row.appendChild(fragment);
   324→  } catch (error) {
   325→    showErrorState(row, error.message, loadTvUpcomingRow);
   326→  }
   327→}

   329→/** Derive the "New episode" / "New season" flag and "S1·E3"
   330→    metadata from a show's next/last episode data. */
   331→function describeTvEpisode(show) {
   332→  const next = show.next_episode_to_air;
   333→  const last = show.last_episode_to_air;
   334→  const today = new Date().toISOString().slice(0, 10);
   335→
   336→  if (next && next.season_number && next.episode_number && next.air_date >= today) {
   337→    return {
   338→      badge: next.episode_number === 1 ? { text: 'New season' } : { text: 'New episode' },
   339→      metaNote: `S${next.season_number}·E${next.episode_number}`
   340→    };
   341→  }
   342→  if (last && last.season_number && last.episode_number) {
   343→    const airedRecently = last.air_date &&
   344→      (Date.now() - new Date(last.air_date).getTime()) / 86400000 <= 7;
   345→    return {
   346→      badge: airedRecently ? { text: 'New episode' } : null,
   347→      metaNote: `S${last.season_number}·E${last.episode_number}`
   348→    };
   349→  }
   350→  return { badge: null, metaNote: null };
   351→}

   489→    const el = $(sel);

   507→            <svg viewBox="0 0 24 24" fill="currentColor"><path d="M17.65 6.35A7.95 7.95 0 0 0 12 4a8 8 0 1 0 7.73 10h-2.08A6 6 0 1 1 12 6c1.66 0 3.14.69 4.22 1.78L13 11h7V4l-2.35 2.35z"/></svg>

   517→  if (retry) {

   530→function renderHeroSlides() {
   531→  const hero = $('#hero');
   532→  if (!hero) return;
   533→
   534→  // Remove skeleton + any stale error UI, restore controls
   535→  const skeleton = $('#hero-skeleton');
   536→  if (skeleton) skeleton.remove();
   537→  const staleError = $('#hero-error');
   538→  if (staleError) staleError.remove();
   539→  // Clear any previously rendered slides (defensive against re-entry)
   540→  $$('.hero-slide', hero).forEach((slide) => slide.remove());
   541→  ['#hero-dots', '#hero-prev', '#hero-next'].forEach((sel) => {
   542→    const el = $(sel);
   543→    if (el) el.removeAttribute('hidden');
   544→  });
   545→
   546→  // Build slides
   547→  const slidesHtml = heroSlides.map((item, index) => {
   548→    const type = getMediaType(item);
   549→    const title = getMediaTitle(item);
   550→    const year = getMediaYear(item);
   551→    const rating = item.vote_average ? item.vote_average.toFixed(1) : 'N/A';
   552→    const backdrop = imageUrl(item.backdrop_path, BACKDROP_SIZE, FALLBACK_STILL);
   553→    const genreNames = (item.genre_ids || [])
   554→      .map((id) => heroGenreMap[id] || GENRE_MAP[id])
   555→      .filter(Boolean)
   556→      .slice(0, 3);
   557→    const overview = item.overview || 'No description available.';
   558→

   646→function goToSlide(index, manual = false) {
   647→  if (!heroSlides.length) return;
   648→  const slides = $$('.hero-slide');
   649→  const dots = $$('.hero-dot');
   650→  if (!slides.length) return;
   651→
   652→  currentSlide = (index + heroSlides.length) % heroSlides.length;
   653→
   654→  slides.forEach((slide, i) => {
   655→    slide.classList.toggle('active', i === currentSlide);
   656→    slide.setAttribute('aria-hidden', String(i !== currentSlide));
   657→  });
   658→  dots.forEach((dot, i) => dot.classList.toggle('active', i === currentSlide));
   659→
   660→  syncHeroTrailer();
   661→
   662→  if (manual) startAutoSlide(); // reset the timer
   663→}
   664→
   665→function startAutoSlide() {
   666→  stopAutoSlide();
   667→  if (!heroSlides.length) return;
   668→  heroTimer = setInterval(() => goToSlide(currentSlide + 1), CONFIG.heroAutoPlayInterval);
   669→}

   671→function stopAutoSlide() {
   672→  if (heroTimer) {
   673→    clearInterval(heroTimer);
   674→    heroTimer = null;
   675→  }

   678→/* ============================================================
   679→   HERO TRAILER (Netflix-style auto-playing muted trailer)
   680→   Every slide keeps an empty .hero-trailer layer between its
   681→   backdrop image and the gradient overlay. While a slide is
   682→   active we: show the backdrop for 2s, fetch the title's
   683→   trailer key (cached), fade a single shared chrome-free
   684→   <video> (direct MP4 sources — VidSrc trailer endpoint, then
   685→   Invidious stream proxies) into that slide's layer, and swap
   686→   it out again when the slide changes. When every source
   687→   fails the backdrop itself gets the Ken Burns pan/zoom, so
   688→   the hero never falls flat.
   689→   ============================================================ */
   690→const HERO_TRAILER_DELAY_MS = 2000; // backdrop-first delay before the video fades in
   691→
   692→function syncHeroTrailer() {
   693→  const token = ++heroTrailerToken;
   694→  stopHeroTrailer();
   695→
   696→  const item = heroSlides[currentSlide];
   697→  if (!item || document.hidden) return;
   698→
   699→  // Ambient lighting: the whole site glows with the active
   700→  // slide's colours (Apple TV+ vibrancy).
   701→  ambientBackdrop.setBase(item);
   702→
   703→  const type = getMediaType(item);
   704→  const delay = new Promise((resolve) => setTimeout(resolve, HERO_TRAILER_DELAY_MS));
   705→  const keyPromise = getTrailerKey(item.id, type);
   706→
   707→  Promise.all([delay, keyPromise]).then(([ , key]) => {
   708→    // Slide changed (or trailer stopped) while we were waiting
   709→    if (token !== heroTrailerToken) return;
   710→    if (heroSlides[currentSlide] !== item || document.hidden) return;
   711→    startHeroTrailer(item, key);
   712→  });
   713→}
   714→
   715→function startHeroTrailer(item, videoKey) {
   716→  const slide = $$('.hero-slide')[currentSlide];
   717→  const layer = slide ? slide.querySelector('.hero-trailer') : null;
   718→  if (!layer) return;
   719→
   720→  const mediaType = getMediaType(item);
   721→  const cached = trailerSourceCache.get(trailerCacheKey(item.id, mediaType));
   722→  // Cached total failure or tripped session circuit breaker ->
   723→  // no network, straight to Ken Burns.
   724→  const sources = (cached && cached.status === 'failed') || !trailerChainEnabled()
   725→    ? []
   726→    : orderedTrailerSources(item, videoKey);
   727→  if (!sources.length) {
   728→    applyHeroKenBurns(slide);
   729→    return;
   730→  }
   731→
   732→  // Baseline: the backdrop is alive (Ken Burns pan) while the
   733→  // stream connects — the video fades in over it on playback.
   734→  applyHeroKenBurns(slide);
   735→
   736→  heroTrailerHandle = mountTrailerVideo(layer, sources, {
   737→    watchdogMs: 5000,
   738→    onSourceDead: noteSourceDead,
   739→    onResolved(src) {
   740→      rememberTrailerSource(item.id, mediaType, src);
   741→      if (src) console.log('HERO TRAILER PLAYING', item.id, src);
   742→    },
   743→    onPlaying() {
   744→      // Slide changed while the stream was connecting
   745→      if (heroSlides[currentSlide] !== item) {
   746→        stopHeroTrailer();
   747→        return;
   748→      }
   749→      // The moving picture replaces the Ken Burns pan.
   750→      removeHeroKenBurns(slide);
   751→      heroTrailerMuted = true;
   752→      updateHeroMuteButton();
   753→      requestAnimationFrame(() => requestAnimationFrame(() => layer.classList.add('playing')));
   754→      const muteBtn = $('#hero-mute-btn');
   755→      if (muteBtn) muteBtn.hidden = false;
   756→    },
   757→    onFailed() {
   758→      // Every direct source failed -> Ken Burns backdrop fallback
   759→      // (already running — just make sure it is on).
   760→      if (heroSlides[currentSlide] !== item) return;
   761→      applyHeroKenBurns(slide);
   762→    }
   763→  });
   764→}
   765→
   766→/** Ken Burns fallback: the slide's backdrop slowly pans/zooms
   767→    (20s loop) — premium even without video. */
   768→function applyHeroKenBurns(slide) {
   769→  const backdrop = slide ? slide.querySelector('.hero-backdrop') : null;
   770→  if (backdrop) backdrop.classList.add('trailer-fallback');
   771→}
   772→
   773→function removeHeroKenBurns(slide) {
   774→  const backdrop = slide ? slide.querySelector('.hero-backdrop') : null;
   775→  if (backdrop) backdrop.classList.remove('trailer-fallback');
   776→}
   777→
   778→function stopHeroTrailer() {
   779→  $$('.hero-slide .hero-trailer').forEach((layer) => layer.classList.remove('playing'));
   780→  $$('.hero-slide .hero-backdrop.trailer-fallback').forEach((bd) => bd.classList.remove('trailer-fallback'));
   781→
   782→  const muteBtn = $('#hero-mute-btn');
   783→  if (muteBtn) muteBtn.hidden = true;
   784→
   785→  if (heroTrailerHandle) {
   786→    // Free the decoder immediately — no fade-out wait needed.
   787→    heroTrailerHandle.destroy();
   788→    heroTrailerHandle = null;
   789→  }
   790→  heroTrailerMuted = true;
   791→}
   792→
   793→function toggleHeroTrailerMute() {
   794→  const video = heroTrailerHandle ? heroTrailerHandle.video : null;
   795→  if (!video) return;
   796→  heroTrailerMuted = !heroTrailerMuted;
   797→  video.muted = heroTrailerMuted;
   798→  if (!heroTrailerMuted) {
   799→    try {
   800→      video.volume = 1;
   801→      const attempt = video.play();
   802→      if (attempt && typeof attempt.catch === 'function') attempt.catch(() => {});
   803→    } catch (playError) { /* unmuted resume is best-effort */ }
   804→  }
   805→  updateHeroMuteButton();
   806→}
   807→
   808→function updateHeroMuteButton() {
   809→  const btn = $('#hero-mute-btn');
   810→  if (!btn) return;
   811→  btn.classList.toggle('muted', heroTrailerMuted);
   812→  btn.setAttribute('aria-pressed', String(!heroTrailerMuted));
   813→  const label = heroTrailerMuted ? 'Unmute trailer' : 'Mute trailer';
   814→  btn.setAttribute('aria-label', label);
   815→  btn.title = label;
   816→}

   818→/* ============================================================
   819→   HORIZONTAL SCROLL ARROWS (all content rows)
   820→   ============================================================ */
   821→function initSectionArrows() {
   822→  $$('.scroll-btn').forEach((button) => {
   823→    const row = $(button.dataset.target);
   824→    if (!row) return;
   825→
   826→    // Scroll by ~85% of visible width
   827→    button.addEventListener('click', () => {
   828→      const amount = row.clientWidth * CONFIG.scrollAmountRatio;
   829→      const direction = button.classList.contains('next') ? 1 : -1;
   830→      row.scrollBy({ left: amount * direction, behavior: 'smooth' });
   831→    });
   832→
   833→    // Enable/disable arrows at row edges
   834→    const updateArrows = () => {
   835→      const prevBtn = row.parentElement.querySelector(`.scroll-btn.prev[data-target="${button.dataset.target}"]`);
   836→      const nextBtn = row.parentElement.querySelector(`.scroll-btn.next[data-target="${button.dataset.target}"]`);
   837→      const maxScroll = row.scrollWidth - row.clientWidth;
   838→      if (prevBtn) prevBtn.disabled = row.scrollLeft <= 4;
   839→      if (nextBtn) nextBtn.disabled = row.scrollLeft >= maxScroll - 4;
   840→    };
   841→
   842→    let ticking = false;
   843→    row.addEventListener('scroll', () => {
   844→      if (!ticking) {
   845→        window.requestAnimationFrame(() => {
   846→          updateArrows();
   847→          ticking = false;
   848→        });
   849→        ticking = true;
   850→      }
   851→    }, { passive: true });
   852→
   853→    // Initial state + after content loads
   854→    updateArrows();
   855→    const afterLoad = () => updateArrows();
   856→    setTimeout(afterLoad, 1500);
   857→    setTimeout(afterLoad, 3500);
   858→    window.addEventListener('resize', debounce(updateArrows, 200), { passive: true });
   859→  });
   860→}
