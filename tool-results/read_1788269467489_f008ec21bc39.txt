     1→/* ============================================================
     2→   ZEUS — party.js
     3→   Client for the ZEUS Watch Party room (party.html), talking to the
     4→   Cloudflare Worker + Durable Object backend (party-worker/).
     5→
     6→   - Backend URL is configurable:
     7→       1. localStorage "zeus_party_backend" (Save button on the join
     8→          screen — paste YOUR deployed worker URL once)
     9→       2. DEFAULT_BACKEND below (the zeus-party worker on
    10→          workers.dev — replace YOUR_WORKERS_SUBDOMAIN after
    11→          `wrangler deploy`)
    12→     Works from BOTH Cloudflare Pages and Vercel deployments — no
    13→     localhost is used anywhere.
    14→   - Rooms: no ?room= -> POST {BACKEND}/api/rooms, then the URL is
    15→     rewritten to party.html?room=ID (url/title params preserved for
    16→     the auto "Set Video")
    17→   - WebSocket: {BACKEND}/ws?room=ID&name=ENCODED_NAME  (https -> wss)
    18→   - Message protocol (JSON over the WebSocket):
    19→       -> { type:"join", name }                 (confirm display name)
    20→       -> { type:"set_media", media:{type,url,title} }   (host only)
    21→       -> { type:"play"|"pause"|"seek", time }
    22→       -> { type:"chat", message, ts }
    23→       <- { type:"state", state, you, serverNow }  (join + every update)
    24→       <- { type:"members", members } / { type:"host", hostId }
    25→       <- { type:"chat", name, message, ts }   / { type:"error", message }
    26→   - Synced playback for YouTube (IFrame API) and direct MP4
    27→   - Live chat + member list + invite link
    28→   - Host rules: the host drives playback (play/pause/seek) and is the
    29→     only one who can set the room video; non-hosts can still paste a
    30→     URL locally (it does NOT change the room media)
    31→   - Drift correction: non-hosts compare their player time with the
    32→     room target time; a difference > 1.2s triggers a local seek
    33→   - Sources that are neither YouTube nor direct .mp4 can't be synced —
    34→     the room still works (chat + manual link paste)
    35→============================================================ */

   187→  function parseYouTubeId(url) {
   188→    if (typeof url !== 'string') return null;
   189→    const patterns = [
   190→      /youtu\.be\/([\w-]{11})(?:[?&#]|$)/,
   191→      /[?&]v=([\w-]{11})(?:[&]|$)/,
   192→      /youtube\.com\/embed\/([\w-]{11})(?:[?&#]|$)/,
   193→      /youtube\.com\/shorts\/([\w-]{11})(?:[?&#]|$)/,
   194→    ];
   195→    for (const p of patterns) {
   196→      const m = url.match(p);
   197→      if (m) return m[1];
   198→    }
   199→    return null;
   200→  }

   202→  function detectMediaType(url) {
   203→    const u = (url || '').trim();
   204→    if (!u) return 'unknown';
   205→    if (parseYouTubeId(u)) return 'youtube';
   206→    if (/\.mp4([?#]|$)/i.test(u)) return 'mp4';
   207→    return 'unknown';
   208→  }

   286→      setConn('connected', 'Connected');
   287→      hideJoinError();

   289→         from the ?name= query param on the WS URL) */
   290→      wsSend({ type: 'join', name: state.displayName });

   295→      let msg;

   301→    ws.onclose = () => {
   302→      state.wsOpen = false;
   303→      stopPing();
   304→      scheduleReconnect();
   305→    };

   307→    ws.onerror = () => { /* onclose follows */ };

   315→  function scheduleReconnect() {
   316→    if (state.reconnectTimer) return;
   317→    const delay = Math.min(RECONNECT_BASE * Math.pow(2, state.reconnectAttempts), RECONNECT_MAX);
   318→    state.reconnectAttempts += 1;
   319→
   320→    /* Never connected on this join? After a few quick attempts, stop
   321→       and let the user fix the backend URL on the join screen. */
   322→    if (!state.everConnected && state.reconnectAttempts > 2) {
   323→      setConn('error', "Can't reach backend");
   324→      rejoinOverlay(
   325→        "Can't reach the party backend at " + BACKEND + '. ' +
   326→        'Check the URL above (your deployed worker, e.g. https://zeus-party.YOUR_SUBDOMAIN.workers.dev) and press Save, then Join again.'
   327→      );
   328→      return;
   329→    }
   330→
   331→    setConn('reconnecting', 'Reconnecting…');
   332→    state.reconnectTimer = setTimeout(() => {
   333→      state.reconnectTimer = null;
   334→      connectSocket();
   335→    }, delay);
   336→  }

   373→  function handleMessage(msg) {
   374→    switch (msg.type) {
   375→      case 'state':   onState(msg); break;
   376→      case 'members': onMembers(msg.members); break;
   377→      case 'host':    onHost(msg.hostId); break;
   378→      case 'chat':    onChat(msg); break;
   379→      case 'error':   toast(msg.message || 'Party error', 'error'); break;
   380→      case 'pong':    break; /* keep-alive reply */
   381→      default:        break;
   382→    }
   383→  }

   385→  function onState(msg) {
   386→    const s = msg.state || {};
   387→    if (msg.you) state.myId = msg.you;
   388→    if (typeof msg.serverNow === 'number') state.clockSkew = msg.serverNow - Date.now();
   389→
   390→    state.hostId = s.hostId || null;
   391→    state.isHost = !!(state.myId && state.hostId === state.myId);
   392→
   393→    state.lastState = {
   394→      paused: !!s.paused,
   395→      time: Number(s.time) || 0,
   396→      updatedAt: Number(s.updatedAt) || 0,
   397→      receivedAt: performance.now(),
   398→    };
   399→
   400→    refreshHostUI();
   401→    if (Array.isArray(s.members)) renderMembers(s.members);
   402→    maybeAutoSetFromUrlParam(s);
   403→
   404→    if (s.media && s.media.url) {
   405→      const roomChanged = state.roomMediaUrl !== s.media.url;
   406→      if (roomChanged) {
   407→        state.roomMediaUrl = s.media.url;
   408→        state.localOverride = false;
   409→        applyMedia(s.media, { time: state.lastState.time, paused: state.lastState.paused, force: true });
   410→      } else if (!state.localOverride && !state.isHost) {
   411→        syncPlayback(); /* drift correction */
   412→      }
   413→    } else if (state.roomMediaUrl || state.currentMedia) {
   414→      state.roomMediaUrl = null;
   415→      state.localOverride = false;
   416→      resetStage();
   417→    }
   418→    updateSyncPill();
   419→  }

   506→  /* ================= player: MP4 ================= */

   508→  function loadMP4(media, opts) {
   509→    const v = els.mp4;
   510→    const url = media.url;
   511→    let resolved = url;
   512→    try { resolved = new URL(url, location.href).href; } catch (err) { /* keep as-is */ }
   513→
   514→    if (v.src !== resolved) {
   515→      v.src = url;
   516→      const onMeta = () => {
   517→        v.removeEventListener('loadedmetadata', onMeta);
   518→        try { if (opts.time > 0.3) v.currentTime = opts.time; } catch (err) {}
   519→        if (!opts.paused) { v.play().catch(showTapOverlay); }
   520→      };
   521→      v.addEventListener('loadedmetadata', onMeta);
   522→      v.load();
   523→    } else {
   524→      try { if (Math.abs(v.currentTime - opts.time) > DRIFT_TOLERANCE) v.currentTime = opts.time; } catch (err) {}
   525→      if (!opts.paused && v.paused) { v.play().catch(showTapOverlay); }
   526→      if (opts.paused && !v.paused) v.pause();
   527→    }
   528→  }

   552→  function ensureYouTubeAPI() {
   553→    if (ytApiPromise) return ytApiPromise;
   554→    ytApiPromise = new Promise((resolve, reject) => {
   555→      if (window.YT && window.YT.Player) { resolve(); return; }
   556→      const prev = window.onYouTubeIframeAPIReady;
   557→      window.onYouTubeIframeAPIReady = () => {
   558→        if (typeof prev === 'function') { try { prev(); } catch (err) {} }
   559→        resolve();
   560→      };
   561→      const s = document.createElement('script');
   562→      s.src = 'https://www.youtube.com/iframe_api';
   563→      s.async = true;
   564→      s.onerror = () => { ytApiPromise = null; reject(new Error('YouTube API failed to load')); };
   565→      document.head.appendChild(s);
   566→      setTimeout(() => {
   567→        if (!(window.YT && window.YT.Player)) { ytApiPromise = null; reject(new Error('YouTube API timeout')); }
   568→      }, 12000);
   569→    });
   570→    return ytApiPromise;
   571→  }

   573→  async function loadYouTube(media, opts) {
   574→    const videoId = parseYouTubeId(media.url);
   575→    if (!videoId) {
   576→      /* safety net if the server type disagrees with the URL */
   577→      applyMedia({ type: 'unknown', url: media.url, title: media.title }, opts);
   578→      return;
   579→    }
   580→    try {
   581→      await ensureYouTubeAPI();
   582→    } catch (err) {
   583→      toast("Couldn't load the YouTube player.", 'error');
   584→      return;
   585→    }
   586→    if (!ytPlayer) {
   587→      ytPendingApply = { videoId, opts };
   588→      ytPlayer = new YT.Player('yt-player', {
   589→        width: '100%',
   590→        height: '100%',
   591→        videoId,
   592→        playerVars: {
   593→          controls: 1,
   594→          rel: 0,
   595→          modestbranding: 1,
   596→          playsinline: 1,
   597→        },
   598→        events: {
   599→          onReady: () => {
   600→            const pending = ytPendingApply;
   601→            ytPendingApply = null;
   602→            if (pending) applyYouTubeVideo(pending.videoId, pending.opts);
   603→          },
   604→          onStateChange: onYouTubeStateChange,
   605→          onError: () => toast("Couldn't load this YouTube video.", 'error'),
   606→        },
   607→      });
   608→    } else {
   609→      applyYouTubeVideo(videoId, opts);
   610→    }
   611→  }
   612→
   613→  function applyYouTubeVideo(videoId, opts) {
   614→    if (!ytPlayer) return;
   615→    const t = Math.max(0, Number(opts.time) || 0);
   616→    let currentVideoId = null;
   617→    try { currentVideoId = ytPlayer.getVideoData() && ytPlayer.getVideoData().video_id; } catch (err) {}
   618→
   619→    if (currentVideoId !== videoId) {
   620→      /* per spec: paused -> cueVideoById, playing -> loadVideoById */
   621→      if (opts.paused) {
   622→        try { ytPlayer.cueVideoById({ videoId, startSeconds: t }); } catch (err) {}
   623→      } else {
   624→        try { ytPlayer.loadVideoById({ videoId, startSeconds: t }); } catch (err) {}
   625→      }
   626→    } else {
   627→      try {
   628→        const cur = ytPlayer.getCurrentTime() || 0;
   629→        if (Math.abs(cur - t) > DRIFT_TOLERANCE) ytPlayer.seekTo(t, true);
   630→      } catch (err) {}
   631→    }
   632→    /* then apply pause/play accordingly */
   633→    try { if (opts.paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo(); } catch (err) {}
   634→  }

   636→  function onYouTubeStateChange(e) {
   637→    const S = window.YT.PlayerState;
   638→    if (e.data === S.PLAYING) {
   639→      hideTapOverlay();
   640→      if (state.isHost) emitHostPlayback('play', ytPlayer.getCurrentTime());
   641→    } else if (e.data === S.PAUSED) {
   642→      if (state.isHost) emitHostPlayback('pause', ytPlayer.getCurrentTime());
   643→    } else if (e.data === S.ENDED) {
   644→      if (state.isHost) emitHostPlayback('pause', ytPlayer.getDuration());
   645→    }
   646→  }

   650→  /* Room target position, extrapolated on the SERVER clock (skew
   651→     corrected) so every client computes the same target. */
   652→  function targetTime() {
   653→    const ls = state.lastState;
   654→    if (!ls) return 0;
   655→    if (ls.paused) return ls.time;
   656→    return ls.time + (Date.now() + state.clockSkew - ls.updatedAt) / 1000;
   657→  }

   659→  /* Non-hosts follow the room: pause/play + seek when drifting. */
   660→  function syncPlayback() {
   661→    if (state.isHost || state.localOverride || !state.currentMedia || !state.lastState) return;
   662→    const paused = state.lastState.paused;
   663→    const target = targetTime();
   664→
   665→    if (state.currentMedia.type === 'mp4') {
   666→      const v = els.mp4;
   667→      try { if (Math.abs(v.currentTime - target) > DRIFT_TOLERANCE) v.currentTime = target; } catch (err) {}
   668→      if (paused && !v.paused) v.pause();
   669→      if (!paused && v.paused) v.play().catch(showTapOverlay);
   670→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   671→      try {
   672→        const cur = ytPlayer.getCurrentTime() || 0;
   673→        if (Math.abs(cur - target) > DRIFT_TOLERANCE) ytPlayer.seekTo(target, true);
   674→        if (paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo();
   675→      } catch (err) {}
   676→    }
   677→  }

   685→  /* periodic drift correction + host YouTube seek detection */
   686→  let lastHostYtTime = null;
   687→  setInterval(() => {
   688→    /* host: YouTube has no "seeked" event — detect time jumps
   689→       (both while playing AND while paused) */
   690→    if (state.isHost && state.currentMedia && state.currentMedia.type === 'youtube' && ytPlayer) {
   691→      try {
   692→        const cur = ytPlayer.getCurrentTime();
   693→        const st = ytPlayer.getPlayerState();
   694→        if (cur != null && st !== window.YT.PlayerState.UNSTARTED) {
   695→          if (lastHostYtTime != null && Math.abs(cur - lastHostYtTime) > 1.8) {
   696→            emitHostPlayback('seek', cur);
   697→          }
   698→          lastHostYtTime = cur;
   699→        } else {
   700→          lastHostYtTime = null;
   701→        }
   702→      } catch (err) { lastHostYtTime = null; }
   703→    } else {
   704→      lastHostYtTime = null;
   705→    }
   706→
   707→    /* non-host: follow the room */
   708→    if (!state.isHost) syncPlayback();
   709→    updateSyncPill();
   710→  }, SYNC_INTERVAL);

   712→  function updateSyncPill() {
   713→    if (!state.roomId) { setSync('waiting', 'Waiting…'); return; }
   714→    if (state.isHost) { setSync('host', 'You control playback'); return; }
   715→    if (!state.currentMedia) { setSync('waiting', 'Waiting for video…'); return; }
   716→    if (state.localOverride) { setSync('local', 'Local (not synced)'); return; }
   717→    const ls = state.lastState;
   718→    if (!ls) { setSync('waiting', 'Syncing…'); return; }
   719→
   720→    let localTime = null;
   721→    let localPaused = true;
   722→    if (state.currentMedia.type === 'mp4') {
   723→      localTime = els.mp4.currentTime || 0;
   724→      localPaused = els.mp4.paused;
   725→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   726→      try { localTime = ytPlayer.getCurrentTime() || 0; localPaused = ytPlayer.getPlayerState() !== window.YT.PlayerState.PLAYING; } catch (err) {}
   727→    }
   728→    if (localTime == null) { setSync('waiting', 'Syncing…'); return; }
   729→
   730→    const drift = Math.abs(localTime - targetTime());
   731→    const pausedMatch = !!ls.paused === !!localPaused;
   732→    if (pausedMatch && drift <= DRIFT_TOLERANCE) {
   733→      setSync('synced', ls.paused ? 'Paused · in sync (' + fmtClock(localTime) + ')' : 'In sync · ' + fmtClock(localTime));
   734→    } else {
   735→      setSync('syncing', 'Syncing… ' + fmtClock(localTime));
   736→    }
   737→  }
