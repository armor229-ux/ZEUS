   161→   161→  function parseYouTubeId(url) {
   162→   162→    if (typeof url !== 'string') return null;
   163→   163→    const patterns = [
   164→   164→      /youtu\.be\/([\w-]{11})(?:[?&#]|$)/,
   165→   165→      /[?&]v=([\w-]{11})(?:[&]|$)/,
   166→   166→      /youtube\.com\/embed\/([\w-]{11})(?:[?&#]|$)/,
   167→   167→      /youtube\.com\/shorts\/([\w-]{11})(?:[?&#]|$)/,
   168→   168→    ];
   169→   169→    for (const p of patterns) {
   170→   170→      const m = url.match(p);
   171→   171→      if (m) return m[1];
   172→   172→    }
   173→   173→    return null;
   174→   174→  }

   176→   176→  function detectMediaType(url) {
   177→   177→    const u = (url || '').trim();
   178→   178→    if (!u) return 'unknown';
   179→   179→    if (parseYouTubeId(u)) return 'youtube';
   180→   180→    if (/\.mp4([?#]|$)/i.test(u)) return 'mp4';
   181→   181→    return 'unknown';
   182→   182→  }

   241→   241→       (when PARTY_SERVER_URL is set, it becomes an absolute URL). */

   293→   293→    refreshHostUI();

   295→   295→    maybeAutoSetFromUrlParam(room);
   296→   296→
   297→   297→    if (room.media && room.media.url) {
   298→   298→      const roomChanged = state.roomMediaUrl !== room.media.url;
   299→   299→      if (roomChanged) {
   300→   300→        state.roomMediaUrl = room.media.url;
   301→   301→        state.localOverride = false;
   302→   302→        applyMedia(room.media, { time: state.lastState.time, paused: state.lastState.paused, force: true });
   303→   303→      } else if (!state.localOverride && !state.isHost) {
   304→   304→        syncPlayback(); /* drift correction */
   305→   305→      }
   306→   306→    } else if (state.roomMediaUrl || state.currentMedia) {
   307→   307→      state.roomMediaUrl = null;
   308→   308→      state.localOverride = false;
   309→   309→      resetStage();
   310→   310→    }
   311→   311→    updateSyncPill();
   312→   312→  }

   344→   344→      state.lastState.time = Number(d && d.time) || 0;

   350→   350→    if (kind === 'seek') { hardSeek(Number(d && d.time) || 0); }
   351→   351→    else { syncPlayback(); }

   425→   425→  /* ================= player: MP4 ================= */
   426→   426→
   427→   427→  function loadMP4(media, opts) {
   428→   428→    const v = els.mp4;
   429→   429→    const url = media.url;
   430→   430→    let resolved = url;
   431→   431→    try { resolved = new URL(url, location.href).href; } catch (err) { /* keep as-is */ }
   432→   432→
   433→   433→    if (v.src !== resolved) {
   434→   434→      v.src = url;
   435→   435→      const onMeta = () => {
   436→   436→        v.removeEventListener('loadedmetadata', onMeta);
   437→   437→        try { if (opts.time > 0.3) v.currentTime = opts.time; } catch (err) {}
   438→   438→        if (!opts.paused) { v.play().catch(showTapOverlay); }
   439→   439→      };
   440→   440→      v.addEventListener('loadedmetadata', onMeta);
   441→   441→      v.load();
   442→   442→    } else {
   443→   443→      try { if (Math.abs(v.currentTime - opts.time) > DRIFT_TOLERANCE) v.currentTime = opts.time; } catch (err) {}
   444→   444→      if (!opts.paused && v.paused) { v.play().catch(showTapOverlay); }

   471→   471→  function ensureYouTubeAPI() {
   472→   472→    if (ytApiPromise) return ytApiPromise;
   473→   473→    ytApiPromise = new Promise((resolve, reject) => {
   474→   474→      if (window.YT && window.YT.Player) { resolve(); return; }

   476→   476→      window.onYouTubeIframeAPIReady = () => {

   478→   478→        resolve();

   481→   481→      s.src = 'https://www.youtube.com/iframe_api';
   482→   482→      s.async = true;
   483→   483→      s.onerror = () => { ytApiPromise = null; reject(new Error('YouTube API failed to load')); };
   484→   484→      document.head.appendChild(s);
   485→   485→      setTimeout(() => {
   486→   486→        if (!(window.YT && window.YT.Player)) { ytApiPromise = null; reject(new Error('YouTube API timeout')); }
   487→   487→      }, 12000);
   488→   488→    });
   489→   489→    return ytApiPromise;

   492→   492→  async function loadYouTube(media, opts) {
   493→   493→    const videoId = parseYouTubeId(media.url);
   494→   494→    if (!videoId) {
   495→   495→      /* safety net if the server type disagrees with the URL */
   496→   496→      applyMedia({ type: 'unknown', url: media.url, title: media.title }, opts);
   497→   497→      return;
   498→   498→    }
   499→   499→    try {
   500→   500→      await ensureYouTubeAPI();
   501→   501→    } catch (err) {
   502→   502→      toast("Couldn't load the YouTube player.", 'error');
   503→   503→      return;
   504→   504→    }
   505→   505→    if (!ytPlayer) {
   506→   506→      ytPendingApply = { videoId, opts };
   507→   507→      ytPlayer = new YT.Player('yt-player', {
   508→   508→        width: '100%',
   509→   509→        height: '100%',
   510→   510→        videoId,
   511→   511→        playerVars: {
   512→   512→          controls: 1,
   513→   513→          rel: 0,
   514→   514→          modestbranding: 1,
   515→   515→          playsinline: 1,
   516→   516→        },
   517→   517→        events: {
   518→   518→          onReady: () => {
   519→   519→            const pending = ytPendingApply;
   520→   520→            ytPendingApply = null;
   521→   521→            if (pending) applyYouTubeVideo(pending.videoId, pending.opts);
   522→   522→          },
   523→   523→          onStateChange: onYouTubeStateChange,
   524→   524→          onError: () => toast("Couldn't load this YouTube video.", 'error'),
   525→   525→        },
   526→   526→      });
   527→   527→    } else {
   528→   528→      applyYouTubeVideo(videoId, opts);
   529→   529→    }
   530→   530→  }

   532→   532→  function applyYouTubeVideo(videoId, opts) {
   533→   533→    if (!ytPlayer) return;
   534→   534→    const t = Math.max(0, Number(opts.time) || 0);
   535→   535→    let currentVideoId = null;
   536→   536→    try { currentVideoId = ytPlayer.getVideoData() && ytPlayer.getVideoData().video_id; } catch (err) {}
   537→   537→
   538→   538→    if (currentVideoId !== videoId) {
   539→   539→      /* per spec: paused -> cueVideoById, playing -> loadVideoById */
   540→   540→      if (opts.paused) {
   541→   541→        try { ytPlayer.cueVideoById({ videoId, startSeconds: t }); } catch (err) {}
   542→   542→      } else {
   543→   543→        try { ytPlayer.loadVideoById({ videoId, startSeconds: t }); } catch (err) {}
   544→   544→      }
   545→   545→    } else {
   546→   546→      try {
   547→   547→        const cur = ytPlayer.getCurrentTime() || 0;
   548→   548→        if (Math.abs(cur - t) > DRIFT_TOLERANCE) ytPlayer.seekTo(t, true);
   549→   549→      } catch (err) {}
   550→   550→    }
   551→   551→    /* then apply pause/play accordingly */
   552→   552→    try { if (opts.paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo(); } catch (err) {}
   553→   553→  }

   562→   562→    } else if (e.data === S.ENDED) {

   564→   564→    }
   565→   565→  }

   591→   591→        if (paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo();

   623→   623→            emitHostPlayback('seek', cur);

   626→   626→        } else {
   627→   627→          lastHostYtTime = null;
   628→   628→        }
   629→   629→      } catch (err) { lastHostYtTime = null; }
   630→   630→    } else {
   631→   631→      lastHostYtTime = null;
   632→   632→    }

   634→   634→    /* non-host: follow the room */
   635→   635→    if (!state.isHost) syncPlayback();

   654→   654→    }

   656→   656→
   657→   657→    const drift = Math.abs(localTime - targetTime());
