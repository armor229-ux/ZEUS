     1→/* ============================================================
     2→   ZEUS — party.js
     3→   Client for the ZEUS Watch Party room (party.html).
     4→
     5→   - Creates/joins a room on the party server (party-server/)
     6→   - Party server URL is configurable: saved to localStorage
     7→     ("zeus_party_server_url", via the Save button on the join screen)
     8→     or the http://localhost:5050 default when ZEUS itself runs on
     9→     localhost; an https page rejects http:// party servers (mixed
    10→     content) before any connection is attempted
    11→   - Synced playback for YouTube (IFrame API) and direct MP4
    12→   - Live chat + member list + invite link
    13→   - Host rules: the host drives playback (play/pause/seek) and is
    14→     the only one who can set the room video; non-hosts can still
    15→     paste a URL locally (it does NOT change the room media)
    16→   - Drift correction: non-hosts compare their player time with the
    17→     room target time; a difference > 1.2s triggers a local seek
    18→   - Sources that are neither YouTube nor direct .mp4 can't be
    19→     synced — the room still works (chat + manual link paste)
    20→============================================================ */

   132→  async function copyText(text) {
   133→    try {
   134→      await navigator.clipboard.writeText(text);
   135→      return true;
   136→    } catch (err) { /* legacy fallback below */ }
   137→    try {
   138→      const helper = document.createElement('textarea');
   139→      helper.value = text;
   140→      helper.setAttribute('readonly', '');
   141→      helper.style.cssText = 'position:fixed;top:-9999px;left:-9999px;opacity:0;';
   142→      document.body.appendChild(helper);
   143→      helper.select();
   144→      helper.setSelectionRange(0, text.length);
   145→      const ok = document.execCommand('copy');
   146→      helper.remove();
   147→      return ok;
   148→    } catch (err) {
   149→      return false;
   150→    }

   184→  function parseYouTubeId(url) {
   185→    if (typeof url !== 'string') return null;
   186→    const patterns = [
   187→      /youtu\.be\/([\w-]{11})(?:[?&#]|$)/,
   188→      /[?&]v=([\w-]{11})(?:[&]|$)/,
   189→      /youtube\.com\/embed\/([\w-]{11})(?:[?&#]|$)/,
   190→      /youtube\.com\/shorts\/([\w-]{11})(?:[?&#]|$)/,
   191→    ];
   192→    for (const p of patterns) {
   193→      const m = url.match(p);
   194→      if (m) return m[1];
   195→    }
   196→    return null;
   197→  }
   198→
   199→  function detectMediaType(url) {
   200→    const u = (url || '').trim();
   201→    if (!u) return 'unknown';
   202→    if (parseYouTubeId(u)) return 'youtube';
   203→    if (/\.mp4([?#]|$)/i.test(u)) return 'mp4';
   204→    return 'unknown';
   205→  }

   207→  function mediaTitleFor(media) {
   208→    if (media.title && media.title.trim()) return media.title.trim();
   209→    if (media.type === 'youtube') return 'YouTube video';
   210→    if (media.type === 'mp4') {
   211→      try { return decodeURIComponent(media.url.split('/').pop().split('?')[0]) || 'MP4 video'; }
   212→      catch (err) { return 'MP4 video'; }
   213→    }
   214→    return 'Custom source';
   215→  }

   262→  function connectSocket() {
   263→    /* Direct connection to the configured party server — always an
   264→       absolute URL (http://localhost:5050 locally, or the deployed
   265→       https:// URL saved from the join screen). */
   266→    if (!PARTY_SERVER_URL) {
   267→      setConn('error', "Can't reach server");

   269→      return;
   270→    }
   271→    const socket = io(PARTY_SERVER_URL, {
   272→      transports: ['websocket', 'polling'],
   273→      forceNew: true,
   274→      reconnection: true,
   275→      reconnectionDelay: 1000,
   276→      reconnectionDelayMax: 5000,
   277→      timeout: 10000,
   278→    });

   280→
   281→    socket.on('connect', () => {
   282→      setConn('connected', 'Connected');
   283→      hideJoinError();

   285→      socket.emit('join', { roomId: state.roomId, displayName: state.displayName });
   286→    });

   288→    socket.on('disconnect', () => setConn('reconnecting', 'Reconnecting…'));
   289→    socket.on('connect_error', () => setConn('error', "Can't reach server"));

   293→    socket.on('members', onMembers);

   295→    socket.on('play', (d) => onPlaybackEvent('play', d));

   310→  function onState(room) {
   311→    state.lastState = {
   312→      roomId: room.roomId,
   313→      hostSocketId: room.hostSocketId,
   314→      media: room.media,
   315→      paused: !!room.paused,
   316→      time: Number(room.time) || 0,
   317→      receivedAt: performance.now(),
   318→    };
   319→    state.hostSocketId = room.hostSocketId;
   320→    state.isHost = !!room.hostSocketId && state.socket && room.hostSocketId === state.socket.id;
   321→    refreshHostUI();
   322→
   323→    maybeAutoSetFromUrlParam(room);
   324→
   325→    if (room.media && room.media.url) {
   326→      const roomChanged = state.roomMediaUrl !== room.media.url;
   327→      if (roomChanged) {
   328→        state.roomMediaUrl = room.media.url;
   329→        state.localOverride = false;
   330→        applyMedia(room.media, { time: state.lastState.time, paused: state.lastState.paused, force: true });
   331→      } else if (!state.localOverride && !state.isHost) {
   332→        syncPlayback(); /* drift correction */
   333→      }
   334→    } else if (state.roomMediaUrl || state.currentMedia) {
   335→      state.roomMediaUrl = null;
   336→      state.localOverride = false;
   337→      resetStage();
   338→    }
   339→    updateSyncPill();
   340→  }

   369→  function onPlaybackEvent(kind, d) {
   370→    if (state.lastState) {
   371→      state.lastState.receivedAt = performance.now();
   372→      state.lastState.time = Number(d && d.time) || 0;
   373→      if (kind === 'play') state.lastState.paused = false;
   374→      if (kind === 'pause') state.lastState.paused = true;
   375→    }
   376→    if (state.isHost) { updateSyncPill(); return; } /* host drives, never follows */
   377→    if (state.localOverride) return;
   378→    if (kind === 'seek') { hardSeek(Number(d && d.time) || 0); }
   379→    else { syncPlayback(); }
   380→    updateSyncPill();
   381→  }

   453→  /* ================= player: MP4 ================= */

   455→  function loadMP4(media, opts) {
   456→    const v = els.mp4;
   457→    const url = media.url;
   458→    let resolved = url;
   459→    try { resolved = new URL(url, location.href).href; } catch (err) { /* keep as-is */ }
   460→
   461→    if (v.src !== resolved) {
   462→      v.src = url;
   463→      const onMeta = () => {
   464→        v.removeEventListener('loadedmetadata', onMeta);
   465→        try { if (opts.time > 0.3) v.currentTime = opts.time; } catch (err) {}
   466→        if (!opts.paused) { v.play().catch(showTapOverlay); }
   467→      };
   468→      v.addEventListener('loadedmetadata', onMeta);
   469→      v.load();
   470→    } else {
   471→      try { if (Math.abs(v.currentTime - opts.time) > DRIFT_TOLERANCE) v.currentTime = opts.time; } catch (err) {}
   472→      if (!opts.paused && v.paused) { v.play().catch(showTapOverlay); }
   473→      if (opts.paused && !v.paused) v.pause();
   474→    }
   475→  }

   499→  function ensureYouTubeAPI() {
   500→    if (ytApiPromise) return ytApiPromise;
   501→    ytApiPromise = new Promise((resolve, reject) => {
   502→      if (window.YT && window.YT.Player) { resolve(); return; }
   503→      const prev = window.onYouTubeIframeAPIReady;
   504→      window.onYouTubeIframeAPIReady = () => {
   505→        if (typeof prev === 'function') { try { prev(); } catch (err) {} }
   506→        resolve();
   507→      };
   508→      const s = document.createElement('script');
   509→      s.src = 'https://www.youtube.com/iframe_api';
   510→      s.async = true;
   511→      s.onerror = () => { ytApiPromise = null; reject(new Error('YouTube API failed to load')); };
   512→      document.head.appendChild(s);
   513→      setTimeout(() => {
   514→        if (!(window.YT && window.YT.Player)) { ytApiPromise = null; reject(new Error('YouTube API timeout')); }
   515→      }, 12000);
   516→    });
   517→    return ytApiPromise;
   518→  }

   520→  async function loadYouTube(media, opts) {
   521→    const videoId = parseYouTubeId(media.url);
   522→    if (!videoId) {
   523→      /* safety net if the server type disagrees with the URL */
   524→      applyMedia({ type: 'unknown', url: media.url, title: media.title }, opts);
   525→      return;
   526→    }
   527→    try {
   528→      await ensureYouTubeAPI();
   529→    } catch (err) {
   530→      toast("Couldn't load the YouTube player.", 'error');
   531→      return;
   532→    }
   533→    if (!ytPlayer) {
   534→      ytPendingApply = { videoId, opts };
   535→      ytPlayer = new YT.Player('yt-player', {
   536→        width: '100%',
   537→        height: '100%',
   538→        videoId,
   539→        playerVars: {
   540→          controls: 1,
   541→          rel: 0,
   542→          modestbranding: 1,
   543→          playsinline: 1,
   544→        },
   545→        events: {
   546→          onReady: () => {
   547→            const pending = ytPendingApply;
   548→            ytPendingApply = null;
   549→            if (pending) applyYouTubeVideo(pending.videoId, pending.opts);
   550→          },
   551→          onStateChange: onYouTubeStateChange,
   552→          onError: () => toast("Couldn't load this YouTube video.", 'error'),
   553→        },
   554→      });
   555→    } else {
   556→      applyYouTubeVideo(videoId, opts);
   557→    }
   558→  }
   559→
   560→  function applyYouTubeVideo(videoId, opts) {
   561→    if (!ytPlayer) return;
   562→    const t = Math.max(0, Number(opts.time) || 0);
   563→    let currentVideoId = null;
   564→    try { currentVideoId = ytPlayer.getVideoData() && ytPlayer.getVideoData().video_id; } catch (err) {}
   565→
   566→    if (currentVideoId !== videoId) {
   567→      /* per spec: paused -> cueVideoById, playing -> loadVideoById */
   568→      if (opts.paused) {
   569→        try { ytPlayer.cueVideoById({ videoId, startSeconds: t }); } catch (err) {}
   570→      } else {
   571→        try { ytPlayer.loadVideoById({ videoId, startSeconds: t }); } catch (err) {}
   572→      }
   573→    } else {
   574→      try {
   575→        const cur = ytPlayer.getCurrentTime() || 0;
   576→        if (Math.abs(cur - t) > DRIFT_TOLERANCE) ytPlayer.seekTo(t, true);
   577→      } catch (err) {}
   578→    }
   579→    /* then apply pause/play accordingly */
   580→    try { if (opts.paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo(); } catch (err) {}
   581→  }

   583→  function onYouTubeStateChange(e) {
   584→    const S = window.YT.PlayerState;
   585→    if (e.data === S.PLAYING) {
   586→      hideTapOverlay();
   587→      if (state.isHost) emitHostPlayback('play', ytPlayer.getCurrentTime());
   588→    } else if (e.data === S.PAUSED) {
   589→      if (state.isHost) emitHostPlayback('pause', ytPlayer.getCurrentTime());
   590→    } else if (e.data === S.ENDED) {
   591→      if (state.isHost) emitHostPlayback('pause', ytPlayer.getDuration());
   592→    }
   593→  }

   597→  function targetTime() {
   598→    const ls = state.lastState;
   599→    if (!ls) return 0;
   600→    if (ls.paused) return ls.time;

   604→  /* Non-hosts follow the room: pause/play + seek when drifting. */
   605→  function syncPlayback() {
   606→    if (state.isHost || state.localOverride || !state.currentMedia || !state.lastState) return;
   607→    const paused = state.lastState.paused;
   608→    const target = targetTime();
   609→
   610→    if (state.currentMedia.type === 'mp4') {
   611→      const v = els.mp4;
   612→      try { if (Math.abs(v.currentTime - target) > DRIFT_TOLERANCE) v.currentTime = target; } catch (err) {}
   613→      if (paused && !v.paused) v.pause();
   614→      if (!paused && v.paused) v.play().catch(showTapOverlay);
   615→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   616→      try {
   617→        const cur = ytPlayer.getCurrentTime() || 0;
   618→        if (Math.abs(cur - target) > DRIFT_TOLERANCE) ytPlayer.seekTo(target, true);
   619→        if (paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo();
   620→      } catch (err) {}
   621→    }
   622→  }

   624→  function hardSeek(time) {
   625→    if (state.localOverride || !state.currentMedia) return;

   628→      try { els.mp4.currentTime = t; } catch (err) {}
   629→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   630→      try { ytPlayer.seekTo(t, true); } catch (err) {}
   631→    }
   632→  }

   636→    if (!state.socket || !state.socket.connected || !state.isHost || !state.roomId) return;

   640→  /* periodic drift correction + host YouTube seek detection */
   641→  let lastHostYtTime = null;
   642→  setInterval(() => {
   643→    /* host: YouTube has no "seeked" event — detect time jumps
   644→       (both while playing AND while paused) */
   645→    if (state.isHost && state.currentMedia && state.currentMedia.type === 'youtube' && ytPlayer) {
   646→      try {
   647→        const cur = ytPlayer.getCurrentTime();
   648→        const st = ytPlayer.getPlayerState();
   649→        if (cur != null && st !== window.YT.PlayerState.UNSTARTED) {
   650→          if (lastHostYtTime != null && Math.abs(cur - lastHostYtTime) > 1.8) {
   651→            emitHostPlayback('seek', cur);
   652→          }
   653→          lastHostYtTime = cur;
   654→        } else {
   655→          lastHostYtTime = null;
   656→        }
   657→      } catch (err) { lastHostYtTime = null; }
   658→    } else {
   659→      lastHostYtTime = null;
   660→    }
   661→
   662→    /* non-host: follow the room */
   663→    if (!state.isHost) syncPlayback();
   664→    updateSyncPill();
   665→  }, SYNC_INTERVAL);

   667→  function updateSyncPill() {
   668→    if (!state.roomId) { setSync('waiting', 'Waiting…'); return; }
   669→    if (state.isHost) { setSync('host', 'You control playback'); return; }
   670→    if (!state.currentMedia) { setSync('waiting', 'Waiting for video…'); return; }
   671→    if (state.localOverride) { setSync('local', 'Local (not synced)'); return; }
   672→    const ls = state.lastState;
   673→    if (!ls) { setSync('waiting', 'Syncing…'); return; }
   674→
   675→    let localTime = null;
   676→    let localPaused = true;
   677→    if (state.currentMedia.type === 'mp4') {
   678→      localTime = els.mp4.currentTime || 0;
   679→      localPaused = els.mp4.paused;
   680→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   681→      try { localTime = ytPlayer.getCurrentTime() || 0; localPaused = ytPlayer.getPlayerState() !== window.YT.PlayerState.PLAYING; } catch (err) {}
   682→    }
   683→    if (localTime == null) { setSync('waiting', 'Syncing…'); return; }
   684→
   685→    const drift = Math.abs(localTime - targetTime());
   686→    const pausedMatch = !!ls.paused === !!localPaused;
   687→    if (pausedMatch && drift <= DRIFT_TOLERANCE) {
   688→      setSync('synced', ls.paused ? 'Paused · in sync (' + fmtClock(localTime) + ')' : 'In sync · ' + fmtClock(localTime));
   689→    } else {
   690→      setSync('syncing', 'Syncing… ' + fmtClock(localTime));
   691→    }
   692→  }
