     1→/* ============================================================
     2→   ZEUS — party.js
     3→   Client for the ZEUS Watch Party room (party.html).
     4→
     5→   - Creates/joins a room on the party server (party-server/)
     6→   - Synced playback for YouTube (IFrame API) and direct MP4
     7→   - Live chat + member list + invite link
     8→   - Host rules: the host drives playback (play/pause/seek) and is
     9→     the only one who can set the room video; non-hosts can still
    10→     paste a URL locally (it does NOT change the room media)
    11→   - Drift correction: non-hosts compare their player time with the
    12→     room target time; a difference > 1.2s triggers a local seek
    13→   - Sources that are neither YouTube nor direct .mp4 can't be
    14→     synced — the room still works (chat + manual link paste)
    15→============================================================ */

   111→  async function copyText(text) {
   112→    try {
   113→      await navigator.clipboard.writeText(text);
   114→      return true;
   115→    } catch (err) { /* legacy fallback below */ }
   116→    try {
   117→      const helper = document.createElement('textarea');
   118→      helper.value = text;
   119→      helper.setAttribute('readonly', '');
   120→      helper.style.cssText = 'position:fixed;top:-9999px;left:-9999px;opacity:0;';
   121→      document.body.appendChild(helper);
   122→      helper.select();
   123→      helper.setSelectionRange(0, text.length);
   124→      const ok = document.execCommand('copy');
   125→      helper.remove();
   126→      return ok;
   127→    } catch (err) {
   128→      return false;
   129→    }

   161→  function parseYouTubeId(url) {
   162→    if (typeof url !== 'string') return null;
   163→    const patterns = [
   164→      /youtu\.be\/([\w-]{11})(?:[?&#]|$)/,
   165→      /[?&]v=([\w-]{11})(?:[&]|$)/,
   166→      /youtube\.com\/embed\/([\w-]{11})(?:[?&#]|$)/,
   167→      /youtube\.com\/shorts\/([\w-]{11})(?:[?&#]|$)/,
   168→    ];
   169→    for (const p of patterns) {
   170→      const m = url.match(p);
   171→      if (m) return m[1];
   172→    }
   173→    return null;
   174→  }

   176→  function detectMediaType(url) {
   177→    const u = (url || '').trim();
   178→    if (!u) return 'unknown';
   179→    if (parseYouTubeId(u)) return 'youtube';
   180→    if (/\.mp4([?#]|$)/i.test(u)) return 'mp4';
   181→    return 'unknown';
   182→  }

   282→  function onState(room) {
   283→    state.lastState = {
   284→      roomId: room.roomId,
   285→      hostSocketId: room.hostSocketId,
   286→      media: room.media,
   287→      paused: !!room.paused,
   288→      time: Number(room.time) || 0,
   289→      receivedAt: performance.now(),
   290→    };
   291→    state.hostSocketId = room.hostSocketId;
   292→    state.isHost = !!room.hostSocketId && state.socket && room.hostSocketId === state.socket.id;
   293→    refreshHostUI();
   294→
   295→    maybeAutoSetFromUrlParam(room);
   296→
   297→    if (room.media && room.media.url) {
   298→      const roomChanged = state.roomMediaUrl !== room.media.url;
   299→      if (roomChanged) {
   300→        state.roomMediaUrl = room.media.url;
   301→        state.localOverride = false;
   302→        applyMedia(room.media, { time: state.lastState.time, paused: state.lastState.paused, force: true });
   303→      } else if (!state.localOverride && !state.isHost) {
   304→        syncPlayback(); /* drift correction */
   305→      }
   306→    } else if (state.roomMediaUrl || state.currentMedia) {
   307→      state.roomMediaUrl = null;
   308→      state.localOverride = false;
   309→      resetStage();
   310→    }
   311→    updateSyncPill();
   312→  }

   342→    if (state.lastState) {

   351→    else { syncPlayback(); }
   352→    updateSyncPill();

   427→  function loadMP4(media, opts) {
   428→    const v = els.mp4;
   429→    const url = media.url;
   430→    let resolved = url;
   431→    try { resolved = new URL(url, location.href).href; } catch (err) { /* keep as-is */ }
   432→
   433→    if (v.src !== resolved) {
   434→      v.src = url;
   435→      const onMeta = () => {
   436→        v.removeEventListener('loadedmetadata', onMeta);
   437→        try { if (opts.time > 0.3) v.currentTime = opts.time; } catch (err) {}
   438→        if (!opts.paused) { v.play().catch(showTapOverlay); }
   439→      };
   440→      v.addEventListener('loadedmetadata', onMeta);
   441→      v.load();
   442→    } else {
   443→      try { if (Math.abs(v.currentTime - opts.time) > DRIFT_TOLERANCE) v.currentTime = opts.time; } catch (err) {}
   444→      if (!opts.paused && v.paused) { v.play().catch(showTapOverlay); }
   445→      if (opts.paused && !v.paused) v.pause();
   446→    }
   447→  }

   471→  function ensureYouTubeAPI() {
   472→    if (ytApiPromise) return ytApiPromise;
   473→    ytApiPromise = new Promise((resolve, reject) => {
   474→      if (window.YT && window.YT.Player) { resolve(); return; }
   475→      const prev = window.onYouTubeIframeAPIReady;
   476→      window.onYouTubeIframeAPIReady = () => {
   477→        if (typeof prev === 'function') { try { prev(); } catch (err) {} }
   478→        resolve();
   479→      };
   480→      const s = document.createElement('script');
   481→      s.src = 'https://www.youtube.com/iframe_api';
   482→      s.async = true;
   483→      s.onerror = () => { ytApiPromise = null; reject(new Error('YouTube API failed to load')); };
   484→      document.head.appendChild(s);
   485→      setTimeout(() => {
   486→        if (!(window.YT && window.YT.Player)) { ytApiPromise = null; reject(new Error('YouTube API timeout')); }
   487→      }, 12000);
   488→    });
   489→    return ytApiPromise;
   490→  }

   492→  async function loadYouTube(media, opts) {
   493→    const videoId = parseYouTubeId(media.url);
   494→    if (!videoId) {
   495→      /* safety net if the server type disagrees with the URL */
   496→      applyMedia({ type: 'unknown', url: media.url, title: media.title }, opts);
   497→      return;
   498→    }
   499→    try {
   500→      await ensureYouTubeAPI();
   501→    } catch (err) {
   502→      toast("Couldn't load the YouTube player.", 'error');
   503→      return;
   504→    }
   505→    if (!ytPlayer) {
   506→      ytPendingApply = { videoId, opts };
   507→      ytPlayer = new YT.Player('yt-player', {
   508→        width: '100%',
   509→        height: '100%',
   510→        videoId,
   511→        playerVars: {
   512→          controls: 1,
   513→          rel: 0,
   514→          modestbranding: 1,
   515→          playsinline: 1,
   516→        },
   517→        events: {
   518→          onReady: () => {
   519→            const pending = ytPendingApply;
   520→            ytPendingApply = null;
   521→            if (pending) applyYouTubeVideo(pending.videoId, pending.opts);
   522→          },
   523→          onStateChange: onYouTubeStateChange,
   524→          onError: () => toast("Couldn't load this YouTube video.", 'error'),
   525→        },
   526→      });
   527→    } else {
   528→      applyYouTubeVideo(videoId, opts);
   529→    }
   530→  }

   532→  function applyYouTubeVideo(videoId, opts) {
   533→    if (!ytPlayer) return;
   534→    const t = Math.max(0, Number(opts.time) || 0);
   535→    let currentVideoId = null;
   536→    try { currentVideoId = ytPlayer.getVideoData() && ytPlayer.getVideoData().video_id; } catch (err) {}
   537→
   538→    if (currentVideoId !== videoId) {
   539→      /* per spec: paused -> cueVideoById, playing -> loadVideoById */
   540→      if (opts.paused) {
   541→        try { ytPlayer.cueVideoById({ videoId, startSeconds: t }); } catch (err) {}
   542→      } else {
   543→        try { ytPlayer.loadVideoById({ videoId, startSeconds: t }); } catch (err) {}
   544→      }
   545→    } else {
   546→      try {
   547→        const cur = ytPlayer.getCurrentTime() || 0;
   548→        if (Math.abs(cur - t) > DRIFT_TOLERANCE) ytPlayer.seekTo(t, true);
   549→      } catch (err) {}
   550→    }
   551→    /* then apply pause/play accordingly */
   552→    try { if (opts.paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo(); } catch (err) {}
   553→  }

   555→  function onYouTubeStateChange(e) {
   556→    const S = window.YT.PlayerState;
   557→    if (e.data === S.PLAYING) {
   558→      hideTapOverlay();
   559→      if (state.isHost) emitHostPlayback('play', ytPlayer.getCurrentTime());
   560→    } else if (e.data === S.PAUSED) {
   561→      if (state.isHost) emitHostPlayback('pause', ytPlayer.getCurrentTime());
   562→    } else if (e.data === S.ENDED) {
   563→      if (state.isHost) emitHostPlayback('pause', ytPlayer.getDuration());
   564→    }
   565→  }

   569→  function targetTime() {
   570→    const ls = state.lastState;
   571→    if (!ls) return 0;
   572→    if (ls.paused) return ls.time;
   573→    return ls.time + (performance.now() - ls.receivedAt) / 1000;

   576→  /* Non-hosts follow the room: pause/play + seek when drifting. */
   577→  function syncPlayback() {
   578→    if (state.isHost || state.localOverride || !state.currentMedia || !state.lastState) return;
   579→    const paused = state.lastState.paused;
   580→    const target = targetTime();
   581→
   582→    if (state.currentMedia.type === 'mp4') {
   583→      const v = els.mp4;
   584→      try { if (Math.abs(v.currentTime - target) > DRIFT_TOLERANCE) v.currentTime = target; } catch (err) {}
   585→      if (paused && !v.paused) v.pause();
   586→      if (!paused && v.paused) v.play().catch(showTapOverlay);
   587→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   588→      try {
   589→        const cur = ytPlayer.getCurrentTime() || 0;
   590→        if (Math.abs(cur - target) > DRIFT_TOLERANCE) ytPlayer.seekTo(target, true);
   591→        if (paused) ytPlayer.pauseVideo(); else ytPlayer.playVideo();
   592→      } catch (err) {}
   593→    }
   594→  }

   596→  function hardSeek(time) {
   597→    if (state.localOverride || !state.currentMedia) return;
   598→    const t = Math.max(0, Number(time) || 0);
   599→    if (state.currentMedia.type === 'mp4') {
   600→      try { els.mp4.currentTime = t; } catch (err) {}
   601→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   602→      try { ytPlayer.seekTo(t, true); } catch (err) {}
   603→    }
   604→  }

   607→  function emitHostPlayback(kind, time) {
   608→    if (!state.socket || !state.socket.connected || !state.isHost || !state.roomId) return;

   612→  /* periodic drift correction + host YouTube seek detection */
   613→  let lastHostYtTime = null;
   614→  setInterval(() => {
   615→    /* host: YouTube has no "seeked" event — detect time jumps
   616→       (both while playing AND while paused) */
   617→    if (state.isHost && state.currentMedia && state.currentMedia.type === 'youtube' && ytPlayer) {
   618→      try {
   619→        const cur = ytPlayer.getCurrentTime();
   620→        const st = ytPlayer.getPlayerState();
   621→        if (cur != null && st !== window.YT.PlayerState.UNSTARTED) {
   622→          if (lastHostYtTime != null && Math.abs(cur - lastHostYtTime) > 1.8) {
   623→            emitHostPlayback('seek', cur);
   624→          }
   625→          lastHostYtTime = cur;
   626→        } else {
   627→          lastHostYtTime = null;
   628→        }
   629→      } catch (err) { lastHostYtTime = null; }
   630→    } else {
   631→      lastHostYtTime = null;
   632→    }
   633→
   634→    /* non-host: follow the room */
   635→    if (!state.isHost) syncPlayback();
   636→    updateSyncPill();
   637→  }, SYNC_INTERVAL);

   639→  function updateSyncPill() {
   640→    if (!state.roomId) { setSync('waiting', 'Waiting…'); return; }
   641→    if (state.isHost) { setSync('host', 'You control playback'); return; }
   642→    if (!state.currentMedia) { setSync('waiting', 'Waiting for video…'); return; }
   643→    if (state.localOverride) { setSync('local', 'Local (not synced)'); return; }
   644→    const ls = state.lastState;
   645→    if (!ls) { setSync('waiting', 'Syncing…'); return; }
   646→
   647→    let localTime = null;
   648→    let localPaused = true;
   649→    if (state.currentMedia.type === 'mp4') {
   650→      localTime = els.mp4.currentTime || 0;
   651→      localPaused = els.mp4.paused;
   652→    } else if (state.currentMedia.type === 'youtube' && ytPlayer) {
   653→      try { localTime = ytPlayer.getCurrentTime() || 0; localPaused = ytPlayer.getPlayerState() !== window.YT.PlayerState.PLAYING; } catch (err) {}
   654→    }
   655→    if (localTime == null) { setSync('waiting', 'Syncing…'); return; }
   656→
   657→    const drift = Math.abs(localTime - targetTime());
   658→    const pausedMatch = !!ls.paused === !!localPaused;
   659→    if (pausedMatch && drift <= DRIFT_TOLERANCE) {
   660→      setSync('synced', ls.paused ? 'Paused · in sync (' + fmtClock(localTime) + ')' : 'In sync · ' + fmtClock(localTime));
   661→    } else {
   662→      setSync('syncing', 'Syncing… ' + fmtClock(localTime));
   663→    }
   664→  }
